cmake_minimum_required(VERSION 3.25)
project(kryvola_wire_conformance LANGUAGES CXX)
find_package(Protobuf CONFIG REQUIRED)
file(GLOB_RECURSE wire_sources CONFIGURE_DEPENDS "../bindings/cpp/*.cc")
list(FILTER wire_sources EXCLUDE REGEX "grpc\\.pb\\.cc$")
add_executable(wire-cpp cpp/main.cpp ${wire_sources})
target_compile_features(wire-cpp PRIVATE cxx_std_20)
target_include_directories(wire-cpp PRIVATE ../bindings/cpp /opt/jsoncons/include)
target_link_libraries(wire-cpp PRIVATE protobuf::libprotobuf)
file(GLOB_RECURSE rpc_sources CONFIGURE_DEPENDS "../bindings/cpp/*.grpc.pb.cc")
add_library(wire-rpc OBJECT ${rpc_sources})
target_compile_features(wire-rpc PRIVATE cxx_std_20)
target_include_directories(wire-rpc PRIVATE ../bindings/cpp /opt/grpc/include)
target_link_libraries(wire-rpc PRIVATE protobuf::libprotobuf)
