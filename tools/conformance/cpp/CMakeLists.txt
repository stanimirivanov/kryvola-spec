cmake_minimum_required(VERSION 3.25)
project(kryvola_canonical_conformance LANGUAGES C CXX)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
set(BLAKE3_SIMD_TYPE none CACHE STRING "Portable test implementation")
set(BUILD_TESTING OFF CACHE BOOL "Only Kryvola tests")
add_subdirectory(/opt/blake3/c blake3 EXCLUDE_FROM_ALL)
add_library(canonical-schema STATIC schema.cpp)
target_compile_features(canonical-schema PRIVATE cxx_std_20)
target_compile_options(canonical-schema PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(canonical-schema SYSTEM PRIVATE /opt/jsoncons/include)
add_executable(canonical-cpp tests.cpp)
target_compile_features(canonical-cpp PRIVATE cxx_std_20)
target_compile_options(canonical-cpp PRIVATE -Wall -Wextra -Wpedantic -Werror -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
target_link_options(canonical-cpp PRIVATE -fsanitize=address,undefined)
target_include_directories(canonical-cpp SYSTEM PRIVATE /opt/jsoncons/include)
target_link_libraries(canonical-cpp PRIVATE canonical-schema BLAKE3::blake3 OpenSSL::Crypto)
